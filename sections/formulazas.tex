\subsection{Lógica - Conjuntos - Bitwise}
$\oplus$ es el xor (o diferencia simétrica de conjuntos).
\begin{enumerate}[1., nosep]
    \item $a|b = a\oplus b + a \& b$
    \item $a\oplus (a\&b) = (a|b)\oplus b$
    \item $b\oplus (a\&b) = (a|b)\oplus a$
    \item $(a\&b)\oplus (a|b) = a\oplus b$
    \item $a + b = a|b + a\&b = a\oplus b + 2(a\&b)$
    \item $a - b = (a\oplus (a\&b))-((a|b)\oplus a) = ((a|b)\oplus b)-((a|b)\oplus a)$
    \item $a - b = (a\oplus (a\&b))-(b\oplus (a\&b)) = ((a|b)\oplus b)-(b\oplus (a\&b))$
\end{enumerate}

Si $p, q$ son predicados, entonces:
\begin{enumerate}[1., nosep]
    \item $(p \vee q) \iff -(-p \wedge -q)$.
    \item $(p \implies q) \iff (-p \vee q)$.
    \item $(p \oplus q) \iff -(p \iff q)$.
\end{enumerate}

\subsection{Teoria de numeros}
Expontente de la potencia mas grande de $p$ primo que divide a $n!$
\newcommand{\piso}[1]{\left\lfloor#1\right\rfloor}
\begin{equation*}
    \nu_p ( n! ) = \sum_{i=1}^{\piso{ \log_p n } } \piso{\frac{n}{ p^i }}
     = \frac{n - S_p(n)}{p - 1},
\end{equation*}
Donde $S_p(n)$ es la suma de dígitos de $n$ en base $p$.

\subsubsection{Números de Bernoulli}

Secuencia de números racionales $B_m$ definidos por la función generatriz exponencial:
\[
\frac{x}{e^x - 1} = \sum_{m=0}^\infty B_m \frac{x^m}{m!}
\]
\[
\begin{array}{c|ccccccccccc}

\textbf{Relación de Recurrencia:}
\[
\sum_{j=0}^m \binom{m+1}{j} B_j = 0 \implies B_m = -\frac{1}{m+1} \sum_{j=0}^{m-1} \binom{m+1}{j} B_j \quad (B_0 = 1)
\]

\textbf{Cálculo mediante Serie Formal de Potencias (Inversión):}

B(x) = \sum_{m=0}^\infty B_m \frac{x^m}{m!} = \left( \sum_{i=0}^\infty \frac{x^i}{(i+1)!} \right)^{-1} \pmod{x^{k+1}}
\]

\subsubsection{Interpolación de Lagrange}

Dado un conjunto de $n+1$ puntos $\{(x_i, y_i)\}_{i=0}^{n}$ de un polinomio $P_n(x)$ de grado $\le n$, el valor de $P_n(X)$ viene dado por:
\[
P_n(X) = \sum_{i=0}^{n} y_i \ell_i(X), \quad \text{donde } \ell_i(X) = \prod_{\substack{j=0 \\ j \neq i}}^{n} \frac{X - x_j}{x_i - x_j}
\]

\algoinfo{N^2}{N}{Para una única consulta $P_n(x)$}


\textbf{Interpolación baricéntrica}
La fórmula baricéntrica de interpolación de Lagrange para el conjunto de nodos $\{(x_i, y_i)\}_{i=0}^{n}$ se define en su primera forma como:
\begin{equation*}
P_n(x) = L(x) \sum_{i=0}^{n} \frac{w_i y_i}{x - x_i}, 
\quad \text{donde} \quad 
w_i = \left( \prod_{\substack{j=0 \\ j \neq i}}^{n} (x_i - x_j) \right)^{-1}
\end{equation*}
con el polinomio nodal:
\begin{equation*}
L(x) = \prod_{k=0}^{n} (x - x_k) = \left( \sum_{i=0}^{n} \frac{w_i}{x - x_i} \right)^{-1}
\end{equation*}

Segunda forma baricéntrica:
\begin{equation*}
    P_n(x) = \left( \sum_{i=0}^{n} \frac{w_i y_i}{x - x_i} \right) \Big/ \left( \sum_{i=0}^{n} \frac{w_i}{x - x_i} \right)
\end{equation*}

\algoinfo{N^2}{N}{Cal $w_i$ en $O(N^2)$ y consulta $P_n(x)$ en $O(N)$}


\textbf{Caso Particular: Puntos Consecutivos}
Si los puntos son $x_i = i$, precalculamos prefijos y sufijos de $pref_i = \prod_{j=0}^{i-1} (X - j)$ y $suf_i = \prod_{j=i+1}^{n} (X - j)$:
\[
P_n(X) = \sum_{i=0}^{n} y_i \cdot \frac{pref_i \cdot suf_i}{ (-1)^{n-i}\, i! \, (n-i)!}
\]

\algoinfo{N}{N}{Para una única consulta $P_n(x)$}

\subsubsection{Potencias de base fija (geométrica)}
Para $p\ne1$:
\[
\sum_{i=a}^b p^i=\frac{p^{b+1}-p^{a}}{1-p}, \hspace{2mm}
\sum_{i=1}^n i\,p^i
=\frac{p-(n+1)p^{n+1}+np^{n+2}}{(1-p)^2}.
\]
\begin{extended}[]
\[
\sum_{i=0}^n p^i=\frac{1-p^{n+1}}{1-p}, \hspace{2mm}
\sum_{i=1}^n p^i=\frac{p(1-p^n)}{1-p},
\]
Si $p=1$, las cuatro sumas son, respectivamente,
$b-a+1$ , $n(n+1)/2$, $n+1$ y $n$.
\end{extended}


\subsubsection{Series infinitas}
Las siguientes fórmulas requieren $|p|<1$:
\[
\sum_{i=0}^{\infty}p^i=\frac{1}{1-p},\qquad
\sum_{i=1}^{\infty}p^i=\frac{p}{1-p},
\]
\[
\sum_{i=1}^{\infty}i\,p^i=\frac{p}{(1-p)^2},\qquad
\sum_{i=1}^{\infty}i^2p^i=\frac{p(1+p)}{(1-p)^3}.
\]

\subsection{Geometría}


\textbf{Polígonos: área y puntos enteros}
Vértices $P_i=(x_i,y_i)$ en orden; índices módulo $n$.
\algoinfo{n}{1}{}
\[
A=\frac12\left|\sum_{i=0}^{n-1}P_i\times P_{i+1}\right|.
\]
Para un polígono simple sin agujeros y con vértices enteros:
\[
B=\sum_{i=0}^{n-1}
\gcd(|x_{i+1}-x_i|,|y_{i+1}-y_i|),
\]
\[
A=I+\frac B2-1,\qquad I=\frac{2A-B+2}{2}.
\]
$B$: puntos enteros del borde; $I$: puntos enteros interiores.
Un segmento entero no degenerado contiene
$\gcd(|\Delta x|,|\Delta y|)+1$ puntos enteros.


\textbf{Triángulos: Herón, radios y trigonometría}
Lados $a,b,c$ opuestos a $\alpha,\beta,\gamma$;
$s=(a+b+c)/2$, área $A>0$, inradio $r$, circunradio $R$.
\[
A=\sqrt{s(s-a)(s-b)(s-c)},\qquad
r=\frac As,\qquad R=\frac{abc}{4A}.
\]
\[
a^2=b^2+c^2-2bc\cos\alpha,\qquad
A=\frac{bc\sin\alpha}{2},
\]
\[
\frac a{\sin\alpha}=\frac b{\sin\beta}
=\frac c{\sin\gamma}=2R.
\]

\textbf{Cuadriláteros: Brahmagupta y Bretschneider}
Cuadrilátero convexo; lados $a,b,c,d$ en orden,
diagonales $p,q$, $s=(a+b+c+d)/2$.
Sean $\alpha,\gamma$ ángulos opuestos y
$T=(s-a)(s-b)(s-c)(s-d)$.
\[
A^2=T-abcd\cos^2\!\left(\frac{\alpha+\gamma}{2}\right)
=T-\frac{(ac+bd)^2-p^2q^2}{4}.
\]
Si es cíclico:
\[
A=\sqrt T,\qquad pq=ac+bd\quad\text{(Ptolomeo)}.
\]

\textbf{Círculos: intersecciones}
Centros $O_1,O_2$, radios $r_1,r_2>0$;
$d=\|O_2-O_1\|>0$. Hay intersección de circunferencias
si $|r_1-r_2|\le d\le r_1+r_2$.
\[
e=\frac{O_2-O_1}{d},\qquad
t=\frac{d^2+r_1^2-r_2^2}{2d},\qquad
h=\sqrt{r_1^2-t^2},
\]
\[
X_\pm=O_1+te\pm he^\perp.
\]
$h=0$: tangencia. Si $d=0$, coinciden si $r_1=r_2$;
en otro caso no se intersectan.


\textbf{Círculos tangentes: Descartes}
Cuatro círculos mutuamente tangentes; curvaturas
$k_i=1/r_i$, negativas para el círculo que encierra a los otros:
\begin{equation*}
    (k_1+k_2+k_3+k_4)^2 = 2\left( k_1^2 +k_2^2 +k_3^2 +k_4^2   \right)
\end{equation*}



\textbf{Volumen de un simplex}
Vértices $P_0,\ldots,P_n\in\mathbb R^d$, $d\ge n$.
Sea $E$ la matriz con filas $P_i-P_0$, $i=1,\ldots,n$:
\[
V_n=\frac{1}{n!}\sqrt{\det(AA^\top)}.
\]
Si $d=n$, se simplifica a $V_n=|\det A|/n!$.


\subsection{Desigualdades}

\begin{description}
    \item[AM-GM-HM:]  Igualdad ssi $x_1 = \dots = x_n$
    $$\frac{n}{\sum_{i=1}^n \frac{1}{x_i}} \le \sqrt[n]{\prod_{i=1}^n x_i} \le \frac{1}{n} \sum_{i=1}^n x_i \quad (x_i > 0)$$
    
    \item[Media Pot:]  $M_r(x) = \left( \frac{1}{n} \sum x_i^r \right)^{1/r}$: Si
    $r < s$, ent, $M_r(x) \le M_s(x)$
    $$M_{-1} = \text{HM} \quad \le \quad  M_0 = \text{GM} \quad \le  \quad M_1 = \text{AM} \quad \le  \quad M_2 = \text{RMS}$$
    
    
    \item[Valor Absoluto / RMS:] Válida para cualquier $x_i \in \mathbb{R}$:
    $$\left\vert{} \frac{1}{n} \sum_{i=1}^n x_i \right\vert{} \le \frac{1}{n} \sum_{i=1}^n \vert{}x_i\vert{} \le \sqrt{\frac{1}{n} \sum_{i=1}^n x_i^2} = \text{RMS}$$
    
    \item[Cauchy-Schwarz:] Igualdad ssi $\exists k \in \mathbb{R} : a_i = k b_i$
    $$\left( \sum_{i=1}^n a_i b_i \right)^2 \le \left( \sum_{i=1}^n a_i^2 \right) \left( \sum_{i=1}^n b_i^2 \right)$$ 
    
    \item[Forma de Engel (Sedrakyan):]  $b_i > 0$
    $$\sum_{i=1}^n \frac{a_i^2}{b_i} \ge \frac{\left( \sum_{i=1}^n a_i \right)^2}{\sum_{i=1}^n b_i} $$
    
    \item[Hölder:] Con $\frac{1}{p} + \frac{1}{q} = 1$ ($p, q > 1$).
    $$\sum |a_i b_i| \le \left( \sum |a_i|^p \right)^{1/p} \left( \sum |b_i|^q \right)^{1/q}$$ 
    
    \item[Minkowski:] $p \ge 1 $
    $$\left( \sum |a_i + b_i|^p \right)^{1/p} \le \left( \sum |a_i|^p \right)^{1/p} + \left( \sum |b_i|^p \right)^{1/p} $$
    
    \item[Reagrupamiento:] Para $a_1 \le \dots \le a_n$ y $b_1 \le \dots \le b_n$: 
    $$\sum_{i=1}^n a_i b_{n-i+1} \le \sum_{i=1}^n a_i b_{\sigma(i)} \le \sum_{i=1}^n a_i b_i$$
    
    \item[Chebyshev:] Si $a_1 \le \dots \le a_n$ y $b_1 \le \dots \le b_n  $ (Orden opuesto, desigualdad se invierte.)
    $$\frac{1}{n} \sum a_i b_i \ge \left( \frac{1}{n} \sum a_i \right) \left( \frac{1}{n} \sum b_i \right)$$ 
   

    \item[Jensen:] Para $f$ convexa, $w_i \ge 0$ y $\sum w_i = 1$
    $$f\left( \sum w_i x_i \right) \le \sum w_i f(x_i)$$
    
    \item[Bernoulli:] Para $x \ge -1$, con $r \ge 1$ o $r \le 0$. Si $0 \le r \le 1$,  desigualdad se invierte
    $$(1 + x)^r \ge 1 + rx$$ 
    
    \item[Karamata:] Si $A \succ B$ ($A$ mayoriza a $B$) y $f$ es convexa 
    $$ \sum_{i=1}^n f(a_i) \ge \sum_{i=1}^n f(b_i)$$
    
    \item[Schur:] Para $x,y,z \ge 0,\  r > 0$:
    $$x^r(x - y)(x - z) + y^r(y - z)(y - x) + z^r(z - x)(z - y) \ge 0$$
\end{description}
